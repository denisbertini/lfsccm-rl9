cmake_minimum_required(VERSION 3.16)
project(lustre_mpc VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_path(SLURM_INCLUDE_DIR NAMES slurm/slurm.h PATHS /usr/include/slurm /opt/slurm/include)
find_library(SLURM_LIBRARY NAMES slurm PATHS /usr/lib64 /usr/lib)
if(NOT SLURM_INCLUDE_DIR OR NOT SLURM_LIBRARY)
    message(WARNING "Slurm headers/libs not found — plugin disabled")
    set(ENABLE_SLURM_PLUGIN OFF)
endif()
pkg_check_modules(LIBURING liburing QUIET)
if(LIBURING_FOUND)
    set(HAVE_LIBURING ON)
endif()
find_package(Threads REQUIRED)
add_library(lmpc_common STATIC common/types.c common/config.c)
target_include_directories(lmpc_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(lmpc_ipc STATIC common/ipc.c)
target_include_directories(lmpc_ipc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(lmpc_ipc PUBLIC lmpc_common Threads::Threads)
add_library(lmpc_ringbuf STATIC cache_daemon/ring_buffer.c cache_daemon/tmp_mgmt.c)
target_include_directories(lmpc_ringbuf PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(lmpc_ringbuf PUBLIC lmpc_common Threads::Threads)
add_library(lmpc_markov STATIC cache_daemon/markov_model.c)
target_include_directories(lmpc_markov PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(lmpc_markov PUBLIC lmpc_ringbuf Threads::Threads)
add_library(lmpc_prefetch STATIC cache_daemon/prefetch_worker.c)
target_include_directories(lmpc_prefetch PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(lmpc_prefetch PUBLIC lmpc_markov Threads::Threads)
if(HAVE_LIBURING)
    target_compile_definitions(lmpc_prefetch PRIVATE HAVE_IO_URING)
    target_include_directories(lmpc_prefetch PRIVATE ${LIBURING_INCLUDE_DIRS})
    target_link_libraries(lmpc_prefetch PRIVATE ${LIBURING_LIBRARIES})
endif()
add_executable(cache_daemon cache_daemon/daemon_main.c)
target_include_directories(cache_daemon PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(cache_daemon PRIVATE lmpc_prefetch lmpc_ipc)
add_library(preadv_hook SHARED preadv_hook/hook.c)
target_include_directories(preadv_hook PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(preadv_hook PRIVATE lmpc_ringbuf lmpc_markov lmpc_ipc dl)
set_target_properties(preadv_hook PROPERTIES PREFIX "" SOVERSION 1 POSITION_INDEPENDENT_CODE ON)
if(ENABLE_SLURM_PLUGIN)
    add_library(bb_lustre_mpc MODULE bb_c/plugin.c bb_c/jobschedule.c)
    target_include_directories(bb_lustre_mpc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${SLURM_INCLUDE_DIR})
    target_link_libraries(bb_lustre_mpc PRIVATE ${SLURM_LIBRARY} dl)
    set_target_properties(bb_lustre_mpc PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugin")
endif()
install(TARGETS cache_daemon RUNTIME DESTINATION bin)
install(TARGETS preadv_hook LIBRARY DESTINATION lib)
if(ENABLE_SLURM_PLUGIN)
    install(TARGETS bb_lustre_mpc LIBRARY DESTINATION lib/slurm)
endif()
